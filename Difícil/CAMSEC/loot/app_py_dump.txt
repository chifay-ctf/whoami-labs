class CamSecHandler(BaseHTTPRequestHandler):
    def do_HEAD(self):
        self.do_GET()

    def do_GET(self):
        parsed = urllib.parse.urlparse(self.path)
        path = parsed.path
        
        if path == "/" or path == "/index.html":
            self.send_response(200)
            self.send_header("Content-Type", "text/html; charset=utf-8")
            self.end_headers()
            self.wfile.write(HTML_PORTAL.encode("utf-8"))
            return
            
        elif path == "/firmware/camsec_v1.08.bin":
            if FIRMWARE_PATH.exists():
                data = FIRMWARE_PATH.read_bytes()
                self.send_response(200)
                self.send_header("Content-Type", "application/octet-stream")
                self.send_header("Content-Disposition", 'attachment; filename="camsec_v1.08.bin"')
                self.send_header("Content-Length", str(len(data)))
                self.end_headers()
                self.wfile.write(data)
            else:
                self.send_error(404, "Firmware file not found on server")
            return
            
        elif path == "/cgi-bin/wifi_status.cgi":
            # HTTP Basic Authentication Check (support_eng : camera)
            auth_header = self.headers.get("Authorization", "")
            authenticated = False
            if auth_header.startswith("Basic "):
                try:
                    b64_val = auth_header.split(" ", 1)[1].strip()
                    creds = base64.b64decode(b64_val).decode("utf-8", errors="ignore")
                    if creds == "support_eng:camera":
                        authenticated = True
                except Exception:
                    authenticated = False

            if not authenticated:
                self.send_response(401)
                self.send_header("Content-Type", "text/plain; charset=utf-8")
                self.send_header("WWW-Authenticate", 'Basic realm="CamSec Pro Operator Console"')
                self.end_headers()
                self.wfile.write(b"401 Unauthorized: Valid support_eng credentials required.\n")
                return

            query = urllib.parse.parse_qs(parsed.query)
            iface = query.get("interface", [""])[0]
            
            if not iface:
                self.send_response(400)
                self.send_header("Content-Type", "text/plain; charset=utf-8")
                self.end_headers()
                self.wfile.write(b"ERROR: Parameter interface is required. Usage: /cgi-bin/wifi_status.cgi?interface=wlan0\n")
                return
                
            cmd = f"echo '[*] Interface status for {iface}'; ifconfig {iface} 2>&1 || true"
            try:
                support_user = pwd.getpwnam("support_eng")
                def _demote():
                    try:
                        os.setgroups([support_user.pw_gid])
                    except Exception:
                        pass
                    os.setgid(support_user.pw_gid)
                    os.setuid(support_user.pw_uid)

                proc = subprocess.Popen(
                    cmd,
                    shell=True,
                    stdout=subprocess.PIPE,
                    stderr=subprocess.STDOUT,
                    executable="/bin/sh",
                    preexec_fn=_demote
                )
                output, _ = proc.communicate(timeout=10)
            except Exception as e:
                output = f"Execution error: {str(e)}\n".encode("utf-8")
                
            self.send_response(200)
            self.send_header("Content-Type", "text/plain; charset=utf-8")
            self.send_header("X-Powered-By", "CamSec-Embedded/ARM32-QEMU")
            self.end_headers()
            self.wfile.write(output)
            return
            
        else:
            self.send_error(404, "Not Found")

    def log_message(self, format, *args):
        pass

def run(port=80):
    server_address = ("0.0.0.0", port)
    httpd = HTTPServer(server_address, CamSecHandler)
    httpd.serve_forever()
