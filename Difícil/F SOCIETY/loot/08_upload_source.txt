    $upload_dir = '/var/www/html/uploads/';

    $file_tmp = $_FILES['file']['tmp_name'];
    $file_ext = strtolower(pathinfo($file_name, PATHINFO_EXTENSION));
    
    // Weak validation (bypasseable)
    $allowed_exts = array('jpg', 'jpeg', 'png', 'gif', 'pdf', 'txt');
    
    // Check extension (can be bypassed with double extension)
    if (in_array($file_ext, $allowed_exts)) {
        $destination = $upload_dir . basename($file_name);
        
        if (move_uploaded_file($file_tmp, $destination)) {
            $success = "File uploaded successfully: " . htmlspecialchars($file_name);
        } else {
            $error = "Error uploading file.";
        }
    } else {
        $error = "Invalid file type. Allowed: jpg, jpeg, png, gif, pdf, txt";
    }
