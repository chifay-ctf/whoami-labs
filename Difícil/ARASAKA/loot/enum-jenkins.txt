#!/bin/bash

echo "[*] Starting ARASAKA Systems..."

# Iniciar SSH
service ssh start
echo "[+] SSH started on port 22"

# Iniciar Apache
service apache2 start
echo "[+] Apache started on port 80"

# Iniciar Jenkins en background como jenkins user
echo "[+] Starting Jenkins on port 8080..."
runuser -u jenkins /opt/jenkins/start-jenkins.sh &

# Esperar a que Jenkins arranque
echo "[*] Waiting for Jenkins (this may take 30-60 seconds)..."
for i in {1..60}; do
    if netstat -tuln 2>/dev/null | grep -q ":8080"; then
        echo "[+] Jenkins started successfully!"
        break
    fi
    sleep 1
done
